# 完整计算流程图：输入 → 模型过程 → 输出

核对日期：2026-10-12。图示项目已有模型的计算依赖；不是所有版本均已启用的声明。正式2专用版本仍缺需求/效率训练及完整迁移证据。蓝色是输入，紫色是拟合模型，白色是条件计算，绿色是输出，橙色是门槛或缺口；虚线表示反馈、验证或尚待核实的跨回合传播。

flowchart TB
  subgraph INPUT["① 输入变量与模型参数"]
    ID["m,h,t,c,i,J：市场、阶段、回合、国家、队伍、竞争集合"]
    X0["期初状态 X0<br/>C0产能、N0人数、Cash0现金、AR0应收、AP0应付<br/>L0长贷、E0权益、Loss0税亏、Book_M0机器、Book_B0厂房<br/>Inv0库存、L_S0短贷、订单状态"]

  end

  CHECK{"② 身份、单位、来源与版本校验<br/>回合/曲线匹配、竞争组完整、输入合法<br/>财务覆盖范围与迁移门槛通过？"}
  STOP["停止完整链或仅保留范围内组件结果<br/>不得用旧参数补造缺项<br/>不得将范围外结果标成最终利润"]
  ID --> CHECK
  X0 --> CHECK
  HIST --> CHECK
  DEC --> CHECK
  GIVEN --> CHECK
  PARAM --> CHECK

  subgraph DEMAND["③ 行业规模、份额与需求"]
    TOTAL["行业规模 T̂_c<br/>T_prev×(1+g_T) 或 T_prev²/T_prev2<br/>尚无独立行业需求回归"]
    RD["有效研发 x_R=R+ρ_R R_prev<br/>研发乘数 f_R=a_R+b_R·(x_R+c_R)^d_R"]
    SCORE["竞争吸引力 u_i,c<br/>价格项 + 两类营销项 + 研发乘数项<br/>分段版本另加高价惩罚"]
    SHARE["竞争组内 softmax<br/>ŝ_i,c=exp(u_i,c)/Σ_j exp(u_j,c)"]
    D["公司需求 D̂_i,c=T̂_c·ŝ_i,c<br/>输出：国别需求、份额"]
    ALT["对照分支<br/>ABS：D̂=exp(α_c+u)，原版研发可用log1p<br/>等份额：D̂=T̂_c/n_J<br/>与主需求分支择一使用，不相加"]
  end
  CHECK -->|通过| TOTAL
  CHECK --> RD
  CHECK --> ALT
  RD --> SCORE --> SHARE --> D
  TOTAL --> D

  subgraph PRODUCTION["④ 效率、产能、生产与交付"]
    EFF["有效运营 x_O=O+ρ_O O_prev<br/>运营乘数 f_O=a_O+b_O·x_O/(c_O+x_O)<br/>效率 η̂=θ·f_O"]
    CAP["产能 C=(1−δ_C)C0+I<br/>机器投资 K_M=p_I·I"]
    Q["产量 q=min(C·a/100,Σ_c D̂_c)<br/>当前零期初库存、零大客户范围"]
    SALES["按π分配供给<br/>S_c=min(D̂_c,剩余供给)<br/>输出：各国交付销量S_c、期末库存"]
    LABOR["班次产量 q1,q2,q3；人数 N=q/η̂<br/>工资 W=(q1+κ2q2+κ3q3)·w/η̂<br/>解聘 F=max((1−δ_N)N0−N,0)·w·σ_F"]
  end
  CHECK --> EFF
  CHECK --> CAP
  D --> Q
  ALT --> Q
  CAP --> Q --> SALES
  Q --> LABOR
  CAP --> LABOR
  EFF --> LABOR

  subgraph OPERATING["⑤ 收入、成本与经营利润"]
    MAT["材料单价 v_eff<br/>旧版 v+Δv 或新版 v·(1+α_v)<br/>同一计算只选一版"]
    COST["收入 V=ΣP_cS_c；运输 Z=Σk_cS_c<br/>材料 B=q·v_eff；营销 M=G+H+M0<br/>其他经营费 H_O=H0+F+R+O"]
    EBITDA["EBITDA=V−B−W−Z−M−H_O"]
    DEP["折旧 Dep_M=d_M·(Book_M0+K_M)<br/>Dep_B=d_B·Book_B0<br/>EBIT=EBITDA−Dep_M−Dep_B"]
  end
  CHECK --> MAT
  MAT --> COST
  Q --> COST
  SALES --> COST
  LABOR --> COST
  COST --> EBITDA --> DEP
  CAP --> DEP

  subgraph FINANCE["⑥ 财务联立：现金、税与利率反馈"]
    WORK["周转资金与期末长贷<br/>AR=V·d_AR/365；AP=B·d_AP/365<br/>L=L0+ΔL"]
    RATE["利率 r_L<br/>外部给定，或风险模型：<br/>r_base+a_r·exp(−b_r·E/A)<br/>或 r_base+a_r·((L+L_S)/E)^b_r"]
    CASH["长贷利息 J_L=(L0+L)·r_L/2<br/>税息前现金 Cash_pre=Cash0+EBITDA−J_L<br/>+(AR0−AR)+(AP−AP0)−K_M−K_B+ΔL"]
    INNER["现金利息与税迭代至收敛<br/>J_C=max((Cash0+Cash)/2,0)·r_C<br/>PBT=EBIT+J_C−J_L；Tax=max(PBT−Loss0,0)·τ<br/>Π=PBT−Tax；Cash=Cash_pre+J_C−Tax"]
    END["期末状态<br/>E=E0+Π；账面资产=期初+投资−折旧<br/>A=Book_M+Book_B+AR+Cash<br/>税亏结转 Loss_end；短贷缺口=max(Cash_min−Cash,0)"]
    VALID{"收敛、资产负债平衡与覆盖范围检查<br/>A=L+AP+E；短贷缺口是否为0？"}
  end
  COST --> WORK
  CHECK --> RATE
  WORK --> CASH
  RATE --> CASH
  EBITDA --> CASH
  CAP --> CASH
  CASH --> INNER
  DEP --> INNER
  INNER --> END --> VALID
  END -. "风险联动版本：用预测E/A或杠杆重新估计利率，外层迭代" .-> RATE
  VALID -->|未收敛或超出覆盖范围| STOP

  subgraph OUTPUT["⑦ 输出、情景比较与后续验证"]
    RESULT["范围内最终条件输出<br/>D̂_c需求、ŝ_c份额、η̂效率、q产量、S_c销量、N人数<br/>V收入、B/W/Z等成本、EBITDA/EBIT<br/>r_L利率、J_L/J_C利息、Tax税、Π税后利润<br/>Cash现金、AR/AP应收应付、L长贷、E权益、A资产、Loss_end税亏"]
    COMPARE["多候选、多情景重复计算<br/>比较范围内Π、现金与失败分支<br/>情景范围不是概率或置信区间"]
    NEXT["下一回合传播<br/>期末状态按规则结转为X0_next<br/>重新读取前景与曲线、设置对手情景<br/>g_D=(ΣD̂_next/ΣD̂_current)−1"]
    HORIZON["正式t至6回合利润小计 ΣΠ_k<br/>未知利润不按0；终局状态保留<br/>完整六回合动态优化尚未完成"]
    EVAL["实际结算后验证冻结预测<br/>y_actual与ŷ比较：MAE、RMSE、WAPE<br/>份额百分点误差、利率基点误差<br/>先验证旧冻结版本，再另建重训版本"]
  end
  VALID -->|通过| RESULT
  D --> RESULT
  EFF --> RESULT
  LABOR --> RESULT
  RESULT --> COMPARE
  RESULT -. "须核验状态结转规则" .-> NEXT
  NEXT -. "更新下一期输入后重复计算" .-> CHECK
  NEXT -. "各期结果完整且在范围内" .-> HORIZON
  RESULT -. "保存预测；等待实际标签" .-> EVAL

本图中的主需求分支使用竞争份额；绝对需求与等份额为对照。材料加法/比例和风险幂/指数按选定版本择一，不能相加或拼接。财务当前覆盖零期初库存、零期初短贷、零大客户等已声明条件；非零库存、完整短贷计息、订单和部分跨回合规则仍有缺口。短贷利率预测不等于已实现完整短贷融资链。
